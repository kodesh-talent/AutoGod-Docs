## 布局与脚本完整示例


```html
// 中英文版本任选一个运行；两段分别创建独立界面。
// 标签表示控件，属性改变外观；可以每次修改一个值，观察界面变化。
var 布局内容 = `
<界面>
    <状态条 />
    <!--使用锚点布局配合滚动布局可以使应用条自带动画-->
    <锚点布局 宽="最大">
        <!--应用标题栏-->
        <应用条布局 宽="最大">
            <!--
            返回:开启返回按钮
            滑动标识:配置滚动布局滚动时的动画标记
            -->
            <应用条 副标题="分割线" 宽="最大" 标识="mAppbar" 标题="Hr"
                滑动标识="滑动|进入时总是可见" 返回="是" />
        </应用条布局>
        <!--滑动布局 在锚点布局中可以配置控件行为是:适配应用条-->
        <滑动布局 宽="最大" 行为="应用条" 高="最大">
            <!--线性布局:被滑动布局包裹，子控件较多时允许向下滑动-->
            <!--内边距:增加内边距[左,上,右,下]-->
            <线性布局 内边距="20,20,20,100" 宽="最大" 高="最大">
                <!--概览-->
                <文本 可选择="是" 文本="概览" 重力="中间" 高="50" />
                <分割线 主题颜色="#C94F4F" 外边距="5" />
                <分割线 主题颜色="#57965C" 外边距="5" />
                <分割线 主题颜色="#3574F0" 外边距="5" />
                <!--设置颜色-->
                <文本 可选择="是" 文本="设置颜色 主题颜色=颜色" 重力="中间" 高="50" />
                <分割线 主题颜色="#C94F4F" />
                <!--设置尺寸-->
                <文本 可选择="是" 文本="设置尺寸 尺寸=尺寸" 重力="中间" 高="50" />
                <!--允许设置小数，达到最小的尺寸 尺寸=0.1 -->
                <分割线 外边距="5" 尺寸="0.1" />
                <分割线 外边距="5" 尺寸="1" />
                <分割线 外边距="5" 尺寸="5" />
                <分割线 外边距="5" 尺寸="10" />
                <!--设置开始边距-->
                <文本 可选择="是" 文本="设置开始边距 开始边距=边距" 重力="中间" 高="50" />
                <分割线 开始边距="100" />
                <!--设置结束边距-->
                <文本 可选择="是" 文本="设置结束边距 结束边距=边距" 重力="中间" 高="50" />
                <分割线 结束边距="100" />
            </线性布局>
        </滑动布局>
    </锚点布局>
</界面>
`;
var 界面 = $界面.解析布局(布局内容);
// 按布局中的标识取得应用条，点击返回按钮时结束当前界面。
界面.找控件("mAppbar").监听返回(function () {
    界面.结束();
});
界面.显示(); // 完成事件绑定后显示界面。
```
