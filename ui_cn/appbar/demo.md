## JS代码


```javascript
var 界面 = $界面.解析布局("example/$ui - 交互界面/01.appbar-应用条/appbar.xml");
//绑定标题栏返回按钮
界面.找控件("appbar").监听返回(function(){
    界面.结束();
});
var 我的标题栏 = 界面.找控件("mAppbar");
//设置返回按钮点击事件
我的标题栏.监听返回(function(){
    吐司("点击了:返回按钮");
});
//设置菜单选项点击事件
我的标题栏.监听菜单(function(文本){
    吐司("点击了:"+文本);
});
//显示界面
界面.显示();
```
