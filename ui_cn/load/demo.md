# 综合案例

- 更新时间:2026-10-22 10:58:41



使用脚本创建界面


```html
// 将布局文字保存在字符串中；每一行用加号连接，符合 ES5 语法。
// 中英文示例任选一套运行。图片资源需提前放到布局所写的路径。
var 布局 = `
<界面>
    <状态条 />
    <!--使用锚点布局配合滚动布局可以使应用条自带动画-->
    <锚点布局 宽="最大">
        <!--应用标题栏-->
        <应用条布局 宽="最大">
            <!--
            返回:开启返回按钮
            滑动标识:配置滚动布局滚动时的动画标记
            -->
            <应用条 副标题="加载" 宽="最大" 标识="mAppbar" 标题="Load" 滑动标识="scroll|enterAlways"
                返回="是" />
        </应用条布局>
        <!--滑动布局 在锚点布局中可以配置控件行为是:适配应用条-->
        <滑动布局 宽="最大" 行为="应用条" 高="最大">
            <!--线性布局:被滑动布局包裹，子控件较多时允许向下滑动-->
            <!--内边距:增加内边距[左,上,右,下]-->
            <线性布局 内边距="20,20,20,100" 宽="最大" 高="最大">
                <!--概览-->
                <文本 可选择="是" 文本="概览" 重力="中间" 高="50" />
                <加载 标识="mLoad" 显示="否" />
                <按钮 标识="mButton" 文本="显示或隐藏加载器" />
                <!--设置主题颜色-->
                <文本 可选择="是" 文本="设置主题颜色 颜色=颜色" 重力="中间" 高="50" />
                <线性布局 宽="最大" 方向="横向">
                    <加载 右外边距="15" 颜色="#57965C" />
                    <加载 右外边距="15" 颜色="#3574F0" />
                    <加载 颜色="#C94F4F" />
                </线性布局>
                <!--设置颜色组合-->
                <文本 可选择="是" 文本="设置颜色组合 颜色组=颜色1,颜色2..." 重力="中间" 高="50" />
                <加载 颜色组="#57965C,#3574F0,#C94F4F,#FFC31F,#F501F5" />
                <!--设置尺寸-->
                <文本 可选择="是" 文本="设置尺寸 尺寸=尺寸" 重力="中间" 高="50" />
                <线性布局 宽="最大" 方向="横向">
                    <加载 右外边距="15" 尺寸="20" />
                    <加载 右外边距="15" 尺寸="25" />
                    <加载 尺寸="30" />
                </线性布局>
                <!--设置显示于隐藏-->
                <文本 可选择="是" 文本="设置显示于隐藏 显示=是、否" 重力="中间" 高="50" />
                <线性布局 宽="最大" 方向="横向">
                    <加载 右外边距="15" 显示="是" 颜色="#57965C" />
                    <!--第二个加载器被隐藏了，所以实际上只能看到两个加载器-->
                    <加载 右外边距="15" 显示="false" 颜色="#3574F0" />
                    <加载 颜色="#C94F4F" />
                </线性布局>
                <文本 可选择="是" 文本="第二个(蓝色的)加载器被隐藏了，所以实际上只能看到两个加载器" 重力="中间"
                    高="50" />
            </线性布局>
        </滑动布局>
    </锚点布局>
</界面>
`;
// 解析字符串得到界面对象，不需要另建布局文件。
var 界面 = $界面.解析布局(布局);
界面.找控件("mAppbar").监听返回(function () {
    界面.结束();
});
// 这个示例只演示显示状态，不执行真实下载或网络请求。
var 正在显示 = false;
var 加载器 = 界面.找控件("mLoad");
界面.找控件("mButton").点击(function () {
    正在显示 = !正在显示;
    if (正在显示) {
        加载器.显示();
    } else {
        加载器.隐藏();
    }
});
// 绑定好事件后再显示界面。
界面.显示();
```

// 显示与隐藏只改变加载动画，不代表后台任务开始或完成。
var loader = ui.id("mLoad");
loader.hide();
// 本页的按钮回调会切换动画，学习时不需要等待或休眠。
